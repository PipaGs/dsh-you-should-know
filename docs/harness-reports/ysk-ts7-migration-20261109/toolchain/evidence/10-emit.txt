// Probe source that mirrors this repository's ESM style: explicit .js
// specifiers on relative imports, top-level const/function, one class.
export const VERSION = '1.0.0';
export function add(a, b) {
    return a + b;
}
export function memo(key, fn) {
    const hit = cache.get(key);
    if (hit !== undefined)
        return hit;
    const value = fn();
    cache.set(key, value);
    return value;
}
const cache = new Map();
export class Counter {
    n = 0;
    step;
    constructor(step = 1) {
        this.step = step;
    }
    inc() {
        this.n += this.step;
        return this.n;
    }
}
export function firstPoint(points) {
    return points[0];
}
export function label(point) {
    return point?.y !== undefined ? `${point.x},${point.y}` : 'none';
}
export async function delay(ms) {
    await new Promise((resolve) => {
        setTimeout(resolve, ms);
    });
}
